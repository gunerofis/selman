function normalize(value){

    return String(value || "")
        .toLocaleLowerCase("tr-TR")
        .replace(/ı/g,"i")
        .replace(/ğ/g,"g")
        .replace(/ü/g,"u")
        .replace(/ş/g,"s")
        .replace(/ö/g,"o")
        .replace(/ç/g,"c")
        .replace(/\s+/g," ")
        .trim();
}


function absoluteURL(url){

    if(!url){
        return "";
    }

    try{
        return new URL(url, window.location.href).href;
    }
    catch(error){
        return "";
    }
}


function cleanURL(url){

    try{
        var u = new URL(url, window.location.href);

        u.search = "";
        u.hash   = "";

        return u.href.replace(/\/+$/,"");
    }
    catch(error){
        return "";
    }
}

function sameStore(url){

    try{
        var host = new URL(url).hostname.replace(/^www\./,"");
        var here = window.location.hostname.replace(/^www\./,"");

        return host === here;
    }

}

    ];

    return excluded.indexOf(t) === -1;
}

function detectPieces(title){

    var t = normalize(title);

    var patterns = [
        /(\d{1,2})\s*parca/,
        /(\d{1,2})\s*aksesuarli/,
        /(\d{1,2})\s*aksesuar/,
        /(\d{1,2})['’]li\s*sumen/,
        /\s(\d{1,2})\s*$/
    ];

    for(var i=0; i<patterns.length; i++){

        var match = t.match(patterns[i]);

        if(!match){
            continue;
        }

        var number = parseInt(match[1], 10);

        if(number >= 2 && number <= 30){
            return number;
        }
    }

    return null;
}

async function fetchDocument(url){

    var response = await fetch(
        absoluteURL(url),
        {
            credentials: "same-origin",
            cache:       "default"
        }
    );

    if(!response.ok){
        throw new Error("HTTP " + response.status + " - " + url);
    }

    var html = await response.text();

    return new DOMParser().parseFromString(html, "text/html");
}


/* =========================================================
   KATEGORİ KARTINDAN ÜRÜN ÇIKAR

   Faprika kategori kartı ürünün adını, linkini,
   fiyatını ve görselini zaten taşıyor. Görsel lazy-load
   olduğu için gerçek adres data-src içindedir; src ise
   yer tutucudur (lazy-logo.png).

   Görsel ve fiyat ürünün KENDİ kartından okunduğu için
   başka bir ürünün görseline bağlanma riski yoktur ve
   ürün başına ayrı detay isteği gerekmez.
   ========================================================= */

function pickCardImage(item){

    var img = item.querySelector("img");

    if(!img){
        return "";
    }

    var candidates = [
        img.getAttribute("data-src"),
        img.getAttribute("data-original"),
        img.getAttribute("data-lazy"),
        img.getAttribute("src")
    ];

    for(var i=0; i<candidates.length; i++){

        var value = candidates[i];

        if(!value){
            continue;
        }

        if(/lazy-logo|placeholder|blank\.|data:image/i.test(value)){
            continue;
        }

        return absoluteURL(value);
    }

    return "";
}


function pickCardPrice(item){

    var el =
        item.querySelector(".prices .price") ||
        item.querySelector(".product-price .price") ||
        item.querySelector(".price");

    if(!el){
        return "";
    }

    var text = (el.textContent || "").replace(/\s+/g," ").trim();

    return /\d/.test(text) ? text : "";
}


function pickCardOldPrice(item){

    var el =
        item.querySelector(".prices .old-price") ||
        item.querySelector(".old-price");

    if(!el){
        return "";
    }

    var text = (el.textContent || "")
        .replace(/\s+/g," ")
        .replace(/%\s*\d+\s*(indirim)?/i,"")
        .trim();

    return /\d/.test(text) ? text : "";
}


function extractProducts(doc){

    var items = Array.from(doc.querySelectorAll(".product-item"));

    /* Tema değişirse tamamen boş kalmasın */
    if(!items.length){
        items = Array.from(doc.querySelectorAll(".product, .product-grid .item"));
    }

    var result = [];
    var seen   = new Set();

    items.forEach(function(item){

        var anchor =
            item.querySelector("h2.name a[href]") ||
            item.querySelector("h2 a[href]") ||
            item.querySelector(".product-name a[href]") ||
            item.querySelector("a[href]");

        if(!anchor){
            return;
        }

        var title = (
            anchor.textContent ||
            anchor.getAttribute("title") ||
            ""
        ).replace(/\s+/g," ").trim();

        if(!isProductTitle(title)){
            return;
        }

        var url = cleanURL(anchor.getAttribute("href"));

        if(!url || seen.has(url) || !sameStore(url)){
            return;
        }

        seen.add(url);

        result.push({
            title:    title,
            url:      url,
            image:    pickCardImage(item),
            price:    pickCardPrice(item),
            oldPrice: pickCardOldPrice(item),
            color:    detectColor(title),
            pieces:   detectPieces(title)
        });
    });

    return result;
}


/* =========================================================
   KATEGORİ + SAYFALAMA
   ========================================================= */

async function categoryProducts(category){

    var all  = [];
    var seen = new Set();

    for(var page=1; page<=CONFIG.MAX_PAGES; page++){

        var url = new URL(category, window.location.origin);

        if(page > 1){
            url.searchParams.set("pagenumber", String(page));
        }

        var doc;

        try{
            doc = await fetchDocument(url.href);
        }
        catch(error){

            if(page === 1){
                throw error;
            }

            break;
        }

        var products = extractProducts(doc);
        var added    = 0;

        products.forEach(function(product){

            if(seen.has(product.url)){
                return;
            }

            seen.add(product.url);
            all.push(product);
            added++;
        });

        if(added === 0){
            break;
        }
    }

    return all;
}


/* =========================================================
   KATALOG + MALZEME

   Katalog dört kategorinin birleşimidir. Kristal ürünler
   "Sümen Takımları" kategorisi altında listelenmediği için
   yalnızca ana kategori okunsaydı Kristal filtresi hiçbir
   zaman sonuç veremezdi.

   Aynı geçişte malzeme kümeleri de kurulur: malzeme yalnızca
   ürün adına bakılarak belirlenmez, ürün adında "deri"
   geçmese bile Deri Sümen Takımları kategorisindeyse deri
   kabul edilir.
   ========================================================= */

function loadCatalog(){

    if(!LOAD_PROMISE){

        LOAD_PROMISE = Promise.allSettled([
            categoryProducts(CONFIG.MAIN_CATEGORY),
            categoryProducts(CONFIG.MATERIAL_CATEGORIES.deri),
            categoryProducts(CONFIG.MATERIAL_CATEGORIES.ahsap),
            categoryProducts(CONFIG.MATERIAL_CATEGORIES.kristal)
        ])
        .then(function(results){

            var keys   = [null, "deri", "ahsap", "kristal"];
            var merged = new Map();

            results.forEach(function(result, index){

                if(result.status !== "fulfilled"){

                    console.warn(
                        "[Sümenini Bul] Kategori okunamadı:",
                        keys[index] || CONFIG.MAIN_CATEGORY,
                        result.reason
                    );

                    return;
                }

                result.value.forEach(function(product){

                    if(keys[index]){
                        MATERIALS[keys[index]].add(product.url);
                    }

                    if(!merged.has(product.url)){
                        merged.set(product.url, product);
                    }
                });
            });

            if(!merged.size){
                throw new Error("Katalog boş döndü.");
            }

            CATALOG = Array.from(merged.values());

            return CATALOG;
        })
        .catch(function(error){

            LOAD_PROMISE = null;

            throw error;
        });
    }

    return LOAD_PROMISE;
}


/* =========================================================
   MALZEME BELİRLE
   ========================================================= */

function productMaterial(product){

    var title = normalize(product.title);

    var inAhsap = MATERIALS.ahsap.has(product.url);
    var inDeri  = MATERIALS.deri.has(product.url);

    if(MATERIALS.kristal.has(product.url)){
        return "kristal";
    }

    if(inAhsap && (inDeri || title.includes("deri"))){
        return "ahsap-deri";
    }

    if(inAhsap){
        return "ahsap";
    }

    if(inDeri){
        return "deri";
    }

    /* Kategori eşleşmesi yoksa ada bak */

    if(title.includes("kristal")){
        return "kristal";
    }

    if(title.includes("ahsap") && title.includes("deri")){
        return "ahsap-deri";
    }

    if(title.includes("ahsap")){
        return "ahsap";
    }

    if(title.includes("deri")){
        return "deri";
    }

    return "";
}

   ÜRÜN DETAYI

   Sadece Hızlı Önizleme için çekilir: açıklama ve
   büyük görsel. Fiyat/ad/kart görseli zaten katalogdan
   gelir, bu yüzden liste açılırken hiç detay isteği
   yapılmaz.

   Not: Bu mağazada ürün sayfalarında Product JSON-LD
   bulunmuyor, veriler DOM'dan okunur.

/*
 * Ürün sayfasının altında "benzer ürünler" listesi vardır ve
 * o kartlar da .prices/.price taşır. Ana ürünün fiyat bloğunda
 * ise .price sınıfı bulunmaz. Bu yüzden sayfa geneline bakan bir
 * seçici, alakasız bir ürünün fiyatını yakalar.
 * Okuma daima ana ürün bloğuna kilitlenir.
 */
function detailScope(doc){

    return doc.querySelector(".product-details-page") ||
           doc.querySelector(".product-essential") ||
           doc;
}


function detailPrice(doc){

    var scope = detailScope(doc);

    var el =
        scope.querySelector('[class*="price-value-"]') ||
        scope.querySelector(".product-price") ||
        scope.querySelector(".prices .price");

    if(!el){
        return "";
    }

    var text = (el.textContent || "").replace(/\s+/g," ").trim();

    if(!/\d/.test(text)){
        return "";
    }

    /* "15.000,00 TL 17.500,00 TL %14 İNDİRİM" -> güncel fiyat */
    var match = text.match(/[\d.,]+\s*(TL|₺|€|\$|EUR|USD)?/i);

    return match ? match[0].trim() : text;
}


function detailImage(doc){

    var og = doc.querySelector('meta[property="og:image"]');

    if(og && og.getAttribute("content")){

        var url = absoluteURL(og.getAttribute("content"));

        if(url && !url.startsWith("data:")){
            return url;
        }
    }

    var img = detailScope(doc).querySelector(".picture img[src], img[src]");

    return img ? absoluteURL(img.getAttribute("src")) : "";
}


function detailDescription(doc){

    var scope = detailScope(doc);

    var el =
        scope.querySelector(".short-description") ||
        scope.querySelector(".full-description");

    var text = el
        ? (el.textContent || "").replace(/\s+/g," ").trim()
        : "";

    if(!text){

        var meta = doc.querySelector('meta[name="description"]');

        text = meta
            ? (meta.getAttribute("content") || "").replace(/\s+/g," ").trim()
            : "";
    }

    return text;
}


function productDetail(product){

    if(!DETAIL_CACHE.has(product.url)){

        var promise = fetchDocument(product.url)
            .then(function(doc){

                var h1 = doc.querySelector("h1");

                var title = h1
                    ? h1.textContent.replace(/\s+/g," ").trim()
                    : product.title;

                return {
                    title:       title,
                    url:         product.url,
                    image:       detailImage(doc) || product.image,
                    price:       detailPrice(doc) || product.price,
                    description: detailDescription(doc),
                    material:    productMaterial(product),
                    color:       detectColor(title) || product.color,
                    pieces:      detectPieces(title) || product.pieces
                };
            })
            .catch(function(error){

                DETAIL_CACHE.delete(product.url);

                throw error;
            });

        DETAIL_CACHE.set(product.url, promise);
    }

    return DETAIL_CACHE.get(product.url);
}

function visible(element){

    if(!element){
        return false;
    }

    var style = getComputedStyle(element);

    return style.display !== "none" && style.visibility !== "hidden";
}


function findGoogleReviews(){

    var selectors = [
        "#guner-reviews-v2",
        "#google-reviews",
        ".google-reviews",
        "[id*='google-review']",
        "[class*='google-review']",
        "[id*='google_reviews']",
        "[class*='google_reviews']",
        "[id*='reviews']",
        ".ti-widget",
        "[class*='trustindex']",
        "[id*='trustindex']",
        "[class*='review-widget']",
        "[id*='review-widget']",
        "[class*='g-review']"
    ];

    var panel = document.getElementById("go-sumen-home-panel");

    for(var i=0; i<selectors.length; i++){

        var elements = [];

        try{
            elements = Array.from(document.querySelectorAll(selectors[i]));
        }
        catch(error){}

        for(var x=0; x<elements.length; x++){

            var element = elements[x];

            if(panel && (panel === element || panel.contains(element))){
                continue;
            }

            if(visible(element)){
                return element;
            }
        }
    }

    var headings = Array.from(document.querySelectorAll("h1,h2,h3,h4,h5"));

    for(var h=0; h<headings.length; h++){

        if(panel && panel.contains(headings[h])){
            continue;
        }

        var text = normalize(headings[h].textContent);

        if(
            text.includes("google yorum") ||
            text.includes("google degerlendirme") ||
            text.includes("musteri yorumlari") ||
            text.includes("musterilerimiz ne diyor")
        ){
            return headings[h].closest("section") || headings[h].parentElement;
        }
    }

    return null;
}


function positionPanel(){

    var panel = document.getElementById("go-sumen-home-panel");

    if(!panel){
        return false;
    }

    var reviews = findGoogleReviews();

    if(!reviews || !reviews.parentElement){
        return false;
    }

    var target = reviews.closest("section") || reviews;

    if(target === panel || panel.contains(target) || target.contains(panel)){
        return false;
    }

    if(target.previousElementSibling !== panel){
        target.insertAdjacentElement("beforebegin", panel);
    }

    panel.setAttribute("data-position","before-google-reviews");

    return true;
}

/* =========================================================
   FİLTRE
   ========================================================= */

function matches(product){

    if(
        STATE.material !== "all" &&
        productMaterial(product) !== STATE.material
    ){
        return false;
    }

    if(
        STATE.color !== "all" &&
        product.color !== STATE.color
    ){
        return false;
    }

    if(STATE.pieces !== "all"){

        var n = product.pieces;

        if(!n){
            return false;
        }

        if(STATE.pieces === "2-6"  && !(n >= 2 && n <= 6)){
            return false;
        }

        if(STATE.pieces === "7-10" && !(n >= 7 && n <= 10)){
            return false;
        }

        if(STATE.pieces === "11+"  && n < 11){
            return false;
        }
    }

    return true;
}

function createCard(product){

    var card = document.createElement("article");

    card.className = "go-product";

    /* Görsel */

    var media = document.createElement("button");

    media.type      = "button";
    media.className = "go-media";
    media.setAttribute("aria-label", product.title + " - hızlı önizleme");

    if(product.image){

        var image = document.createElement("img");

        image.src     = product.image;
        image.alt     = product.title;
        image.loading = "lazy";

        image.onerror = function(){
            media.innerHTML =
                '<div class="go-placeholder">Ürün görseli yüklenemedi.</div>';
        };

        media.appendChild(image);
    }
    else{
        media.innerHTML =
            '<div class="go-placeholder">Ürün görseli bulunamadı.</div>';
    }

    /* Body */

    var body = document.createElement("div");

    body.className = "go-product-body";

    var tags = document.createElement("div");

    tags.className = "go-tags";

    addTag(tags, materialLabel(productMaterial(product)));
    addTag(tags, colorLabel(product.color));

    if(product.pieces){
        addTag(tags, product.pieces + " Parça");
    }

    var title = document.createElement("a");

    title.className   = "go-name";
    title.href        = product.url;
    title.textContent = product.title;

    var price = document.createElement("div");

    price.className = "go-price";

    if(product.price){

        price.textContent = product.price;

        if(product.oldPrice){

            var old = document.createElement("span");

            old.className   = "go-old";
            old.textContent = product.oldPrice;

            price.appendChild(old);
        }
    }
    else{
        price.textContent = "Fiyat için ürünü incele";
    }

    /* Butonlar */

    var actions = document.createElement("div");

    actions.className = "go-card-actions";

    var preview = document.createElement("button");

    preview.type        = "button";
    preview.className   = "go-preview";
    preview.textContent = "Hızlı Önizle";

    var link = document.createElement("a");

    link.className   = "go-product-link";
    link.href        = product.url;
    link.textContent = "Ürünü İncele";

    actions.appendChild(preview);
    actions.appendChild(link);

    body.appendChild(tags);
    body.appendChild(title);
    body.appendChild(price);
    body.appendChild(actions);

    card.appendChild(media);
    card.appendChild(body);

    media.addEventListener("click", function(){
        openPreview(product);
    });

    preview.addEventListener("click", function(){
        openPreview(product);
    });

    return card;
}

async function showResults(){

    var panel   = document.getElementById("go-sumen-home-panel");
    var status  = panel.querySelector(".go-status");
    var results = panel.querySelector(".go-results");
    var grid    = panel.querySelector(".go-grid");
    var info    = panel.querySelector(".go-result-info");

    grid.innerHTML = "";

    results.classList.add("go-show");
    status.classList.add("go-show");

    status.textContent = "Ürün kataloğu okunuyor...";

    try{

        await loadCatalog();

        var matching = CATALOG.filter(matches);

        info.textContent =
            CATALOG.length + " ürün arasından " +
            matching.length + " eşleşme bulundu.";

        status.classList.remove("go-show");

        if(!matching.length){

            grid.innerHTML =
                '<div class="go-empty">' +
                    'Seçimlerinle eşleşen ürün bulunamadı.<br><br>' +
                    'Başka bir renk, malzeme veya parça sayısı seç.' +
                '</div>';

            return;
        }

        matching
            .slice(0, CONFIG.MAX_RESULTS)
            .forEach(function(product){
                grid.appendChild(createCard(product));
            });
    }
    catch(error){

        console.error("[Sümenini Bul]", error);

        status.classList.add("go-show");

        status.textContent = "Ürün kataloğu yüklenemedi. Lütfen tekrar dene.";
    }
}

    link.href               = product.url;

    tags.innerHTML = "";

    addTag(tags, materialLabel(productMaterial(product)));
    addTag(tags, colorLabel(product.color));

    if(product.pieces){
        addTag(tags, product.pieces + " Parça");
    }

    media.innerHTML = "";

    if(product.image){

        var quick = document.createElement("img");

        quick.src = product.image;
        quick.alt = product.title;

        media.appendChild(quick);
    }
    else{
        media.textContent = "Ürün yükleniyor...";
    }

    modal.classList.add("go-open");
    modal.setAttribute("aria-hidden","false");

    document.body.style.overflow = "hidden";

    /* Detay: açıklama ve büyük görsel */

    try{

        var detail = await productDetail(product);

        if(!modal.classList.contains("go-open")){
            return;
        }

        title.textContent       = detail.title;
        price.textContent       = detail.price || "";
        description.textContent = detail.description
            ? detail.description.slice(0, 430)
            : "";
        link.href               = detail.url;

        tags.innerHTML = "";

        addTag(tags, materialLabel(detail.material));
        addTag(tags, colorLabel(detail.color));

        if(detail.pieces){
            addTag(tags, detail.pieces + " Parça");
        }

        if(detail.image){

            media.innerHTML = "";

            var image = document.createElement("img");

            image.src = detail.image;
            image.alt = detail.title;

            media.appendChild(image);
        }
    }
    catch(error){

        console.warn("[Sümenini Bul] Önizleme detayı alınamadı:", product.url, error);

        description.textContent = "";
    }
}


function closePreview(){

    var panel = document.getElementById("go-sumen-home-panel");

    if(!panel){
        return;
    }

    var modal = panel.querySelector(".go-modal");

    modal.classList.remove("go-open");
    modal.setAttribute("aria-hidden","true");

    document.body.style.overflow = "";
}


/* =========================================================
   RESET
   ========================================================= */

function resetFilters(){

    STATE = {
        material: "all",
        color:    "all",
        pieces:   "all"
    };

    var panel = document.getElementById("go-sumen-home-panel");

    panel.querySelectorAll("[data-group]").forEach(function(group){

        group.querySelectorAll(".go-option").forEach(function(button){
            button.classList.remove("go-active");
        });

        var all = group.querySelector('[data-value="all"]');

        if(all){
            all.classList.add("go-active");
        }
    });

    panel.querySelector(".go-results").classList.remove("go-show");
    panel.querySelector(".go-status").classList.remove("go-show");
}


/* =========================================================
   EVENTLER
   ========================================================= */

function bindEvents(){

    var toggle = document.getElementById("go-sumen-toggle");
    var panel  = document.getElementById("go-sumen-home-panel");

    toggle.addEventListener("click", function(){

        setPanelOpen(!panel.classList.contains("go-open"));
    });

    panel.querySelector(".go-panel-close")
        .addEventListener("click", function(){
            setPanelOpen(false);
        });

    panel.addEventListener("click", function(event){

        var option = event.target.closest(".go-option");

        if(!option){
            return;
        }

        var group = option.closest("[data-group]");

        if(!group){
            return;
        }

        group.querySelectorAll(".go-option").forEach(function(button){
            button.classList.remove("go-active");
        });

        option.classList.add("go-active");

        STATE[group.getAttribute("data-group")] =
            option.getAttribute("data-value");
    });

    panel.querySelector(".go-find")
        .addEventListener("click", showResults);

    panel.querySelector(".go-reset")
        .addEventListener("click", resetFilters);

    panel.querySelector(".go-modal-close")
        .addEventListener("click", closePreview);

    panel.querySelector(".go-modal")
        .addEventListener("click", function(event){

            if(event.target === this){
                closePreview();
            }
        });

    document.addEventListener("keydown", function(event){

        if(event.key === "Escape"){
            closePreview();
        }
    });
}


/* =========================================================
   BAŞLAT
   ========================================================= */

function install(){

    createTopButton();
    createPanel();
    bindEvents();

    /* Katalog arka planda hazırlanır, görseller kart verisiyle gelir */

    var warm = function(){
        loadCatalog().catch(function(){});
    };

    if("requestIdleCallback" in window){
        requestIdleCallback(warm);
    }
    else{
        setTimeout(warm, 1200);
    }
}


function start(){
    install();
}


if(document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded", start);
}
else{
    start();
}

})();
